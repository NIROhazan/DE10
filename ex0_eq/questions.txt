# ============================================================
#  תרגיל 0 - השאלות ש-run.bat שואל, לפי הסדר
#  (קובץ למרצה - הסטודנט לא עורך אותו, רק מריץ run.bat)
# ============================================================
#
#  Qn:  Y = <משוואה> | SW: 0 5 8       חלק 1 - run.bat צורב את המשוואה, הסטודנט מרים את המתגים
#                                       ומקליד מה מוצג ב-HEX3 HEX2 HEX1 HEX0
#  Qn:  Y = <משוואה> | HEX: 7 0 3 4    חלק 2 - הסטודנט מקליד אילו מתגים הרים כדי שהלוח יראה את זה
#                                       (? = כל ספרה; כל תשובה שנותנת את התצוגה מתקבלת)
#  לכל שאלה חייב להיות פתרון שמגיעים אליו עם המתגים והתצוגה - בודקים לפני שמוסיפים שאלה.
#
#  התשובה הנכונה מחושבת מהשורה עצמה - אפשר לשנות/להוסיף שאלות בחופשיות.
#  משוואה חדשה מקומפלת בפעם הראשונה (כדקה) ונשמרת ב-sof/.
#  + = OR, * = AND, ~ או ' = NOT, ^ = XOR, ביט אחרי ביט על 4 ביטים.
# ============================================================

# ------------------------------------------------------------
#  10 שאלות, מהקלה ביותר (1) לקשה ביותר (10)
# ------------------------------------------------------------

# 1 - קל מאוד: רק לקרוא את התצוגה. Y הוא פשוט A
Q1:  Y = A                | SW: 0 2

# 2 - קל: OR ראשון, בלי ביטים משותפים - נראה כמו חיבור
Q2:  Y = A + B            | SW: 1 4

# 3 - קל-בינוני: AND ביט אחרי ביט (7 AND 6 = 6, לא 42)
Q3:  Y = A * B            | SW: 0 1 2 5 6

# 4 - בינוני: בכיוון ההפוך - מהתצוגה למתגים, דרך NOT של 4 ביטים
Q4:  Y = ~A               | HEX: 9 0 0 6

# 5 - בינוני: XOR, צריך לכתוב את A ו-B בבינארי
Q5:  Y = A ^ B            | SW: 0 2 3 4 5

# 6 - בינוני-קשה: סדר פעולות (NOT, אחר כך AND, אחר כך OR) על שלושה משתנים
Q6:  Y = A'B + C          | SW: 0 5 6 7

# 7 - קשה: למצוא את B ו-C כשרק Y ו-A ידועים (חוק הפילוג: (A+B)(A+C) = A+BC)
Q7:  Y = (A + B)(A + C)   | HEX: F ? ? 8

# 8 - קשה: לפתור משוואת XOR - A לא נתון, צריך לחשב אותו מ-Y, B, C
Q8:  Y = A ^ B ^ C        | HEX: 0 5 3 ?

# 9 - מומחה: C' מדליק תמיד את הביט העליון, ו-A לא משפיע על Y בכלל. צריך לפתור ביט-ביט מה C חייב להיות
#     (יש שני פתרונות: C = 2 או C = 6 - שניהם מתקבלים)
Q9:  Y = B + C'           | HEX: D ? 4 A

# 10 - מומחה: לזהות ש-(A+B)(A'+C) הוא בורר (MUX) לכל ביט - איפה ש-A=1 נלקח הביט של C, איפה ש-A=0 נלקח הביט של B
#      (B = 6 או B = 7 - הביט הנמוך של B לא נבחר כי A0 = 1)
Q10: Y = (A + B)(A' + C)  | HEX: 6 4 ? 9
