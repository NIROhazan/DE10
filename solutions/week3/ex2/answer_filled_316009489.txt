# 1. SOP קנוני: סכום של מינטרמים, מינטרם אחד לכל שורה שבה Y=1
SOP = CB'A' + C'BA

# 2. POS קנוני: מכפלה של מקסטרמים, מקסטרם אחד לכל שורה שבה Y=0
POS = (C'+B+A')(C'+B'+A)(C'+B'+A')(C+B+A)(C+B+A')(C+B'+A)

# 3. POS מינימלי: הכי מעט סוגריים שאתם מצליחים, עדיין צורת POS (מכפלה של סכומים)
MIN = (C+B)(B'+A)(A'+C')

#  שאלות - ענו במילים שלכם, משפט או שניים

# ש1. בתרגיל 1 ה-POS היה קצר מה-SOP. כאן זה הפוך. מה בטבלה קובע איזו צורה קנונית קצרה יותר?
ת1 = the number of 1s vs 0s: SOP has one term per 1 (2 here), POS one per 0 (6 here)

# ש2. אפשר לאחד את שני המינטרמים של ה-SOP לאיבר אחד קצר יותר? הסבירו למה כן או למה לא.
ת2 = no - terms merge only if they differ in exactly one literal; 000 and 111 differ in three

# ש3. תארו במילים, בלי נוסחה, מתי Y=1.
ת3 = Y = 1 when all three inputs are equal

# ש4. נבאו לפני שאתם מזיזים מתגים: A=0 B=1 C=0 - אילו נורות LEDR יידלקו?
ת4 = A=0 B=1 C=0 -> LEDR0, LEDR1, LEDR2, LEDR9 all off (Y = 0)
