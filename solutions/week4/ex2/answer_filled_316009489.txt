# 1. SOP קנוני: סכום של מינטרמים, מינטרם אחד לכל שורה שבה Y=1
SOP = C'D'A'B' + C'D'A'B + C'D'AB' + C'D'AB + CD'A'B' + CD'AB' + CDAB'

# 2. POS קנוני: מכפלה של מקסטרמים, מקסטרם אחד לכל שורה שבה Y=0
POS = (C+D'+A+B)(C+D'+A+B')(C+D'+A'+B)(C+D'+A'+B')(C'+D+A+B')(C'+D+A'+B')(C'+D'+A+B)(C'+D'+A+B')(C'+D'+A'+B')

# 3. SOP מינימלי: ציירו מפת קרנו של ה-1, הכי מעט איברים ובהם הכי מעט ליטרלים
MSOP = C'D' + D'B' + CAB'

# 4. POS מינימלי: ציירו מפת קרנו של ה-0, הכי מעט סוגריים ובהם הכי מעט ליטרלים
MPOS = (C+D')(C'+B')(D'+A)

# ש1. יש במפה קבוצה של ארבעה 1 שלא נראים צמודים זה לזה בציור. איפה הם, ומה הופך אותם לשכנים?
ת1 = the four corners 0000, 0010, 1000, 1010: the map wraps left-right and top-bottom

# ש2. לכל איבר ב-MSOP שלכם: ציינו שורה אחת (ABCD) שרק הוא מכסה. אם אין כזו - מה זה אומר על האיבר?
ת2 = A'B' only covers 0001 (or 0011); B'D' only 1000; ACD' only 1110 -> all three essential

# ש3. בשבוע 3 (תרגילים 1-3) ראינו "פחות 1 -> SOP קנוני קצר". כאן יש 7 אחדים ו-9 אפסים.
#     ספרו ליטרלים ב-MSOP וב-MPOS שלכם. מי ניצח, ולמה הכלל על הקנוניים לא מנבא את המינימליים?
ת3 = MSOP 7 literals, MPOS 6 -> POS wins; 1s/0s count fixes the canonical length, the minimal

# ש4. נבאו לפני שאתם נוגעים בלוח: A=1 B=0 C=1 D=1 (D הוא KEY3, משוחרר = 0) - מה יהיה LEDG0, ואילו נורות LEDR יידלקו?
ת4 = A=1 B=0 C=1 D=1 -> 1110, Y = 1: LEDG0 on, LEDR0-3 on, LEDR9 off
